     1→     1→"""
     2→     2→In-memory network state — the bridge between SQLite and the DSA modules.
     3→     3→
     4→     4→What lives in memory (loaded once at startup)
     5→     5→--------------------------------------------
     6→     6→    users         hash table  username -> User          (O(1) lookups)
     7→     7→    graph         SocialGraph adjacency lists            (accepted + pending edges)
     8→     8→    trie          Trie                                  (prefix search index)
     9→     9→    posts         list + hash table id -> Post
    10→    10→    likes         hash table  post_id -> set(usernames) (O(1) "did I like this?")
    11→    11→    comments      hash table  post_id -> chronological list
    12→    12→    requests      hash table  request_id -> ConnectionRequest (the pending queue)
    13→    13→    notifications hash table  username -> FIFO list
    14→    14→
    15→    15→Every mutation updates the in-memory structures FIRST, then mirrors the
    16→    16→change into SQLite — so the database is pure persistence while ALL
    17→    17→computation (BFS paths, mutuals, feed ordering, recommendations, search)
    18→    18→happens on these in-memory data structures.
    19→    19→
    20→    20→A re-entrant lock guards all operations because the Flask dev server is

    66→    66→def clean_list(value) -> List[str]:
    67→    67→    """Normalise skills / interests coming from JSON: accepts a list or a
    68→    68→    comma-separated string; deduplicates and trims."""
    69→    69→    if isinstance(value, str):
    70→    70→        parts = value.split(",")
    71→    71→    elif isinstance(value, (list, tuple)):
    72→    72→        parts = [str(item) for item in value]
    73→    73→    else:
    74→    74→        parts = []
    75→    75→    cleaned: List[str] = []
    76→    76→    for part in parts:
    77→    77→        part = part.strip()
    78→    78→        if part and len(part) <= 40 and part not in cleaned:
    79→    79→            cleaned.append(part)

   103→   103→    def initialize(self) -> None:
   104→   104→        """Create tables, seed sample data on first boot, load memory."""
   105→   105→        with self._lock:

   111→   111→    def reload_from_db(self) -> None:
   112→   112→        """Rebuild every in-memory structure from SQLite. This is the only
   113→   113→        place the database is read in bulk — graph operations afterwards
   114→   114→        never touch SQL."""
   115→   115→        with self._lock:
   116→   116→            self.users = {}
   117→   117→            self.graph = SocialGraph()
   118→   118→            self.trie = Trie()
   119→   119→            self.posts, self.posts_by_id = [], {}
   120→   120→            self.likes, self.comments = {}, {}
   121→   121→            self.requests = {}
   122→   122→            self.notifications = {}
   123→   123→            self._post_counts = {}
   124→   124→            self._indexed_keywords = {}
   125→   125→
   126→   126→            for user in db.fetch_all_users():
   127→   127→                self.users[user.username] = user
   128→   128→                self.graph.add_vertex(user.username)
   129→   129→                self._index_user(user)
   130→   130→
   131→   131→            for user_a, user_b in db.fetch_connections():
   132→   132→                if user_a in self.users and user_b in self.users \
   133→   133→                        and not self.graph.is_connected(user_a, user_b):
   134→   134→                    self.graph.add_edge(user_a, user_b)
   135→   135→
   136→   136→            for request in db.fetch_pending_requests():
   137→   137→                self.requests[request.id] = request
   138→   138→                self.graph.add_pending(request.sender, request.receiver)
   139→   139→
   140→   140→            for post in db.fetch_all_posts():
   141→   141→                self.posts.append(post)
   142→   142→                self.posts_by_id[post.id] = post
   143→   143→                self._post_counts[post.author] = self._post_counts.get(post.author, 0) + 1
   144→   144→
   145→   145→            self.likes = db.fetch_likes()
   146→   146→            self.comments = db.fetch_comments()
   147→   147→
   148→   148→            for note in db.fetch_all_notifications():
   149→   149→                self.notifications.setdefault(note.recipient, []).append(note)

   154→   154→    def _index_user(self, user: User) -> None:
   155→   155→        """Insert a student's searchable keywords (username + name words)
   156→   156→        into the Trie."""
   157→   157→        keywords = [user.username] + user.full_name.split()
   158→   158→        for keyword in keywords:
   159→   159→            self.trie.insert(keyword, user.username)
   160→   160→        self._indexed_keywords[user.username] = keywords
   161→   161→
   162→   162→    def _reindex_user(self, user: User) -> None:
   163→   163→        """Remove the old Trie keywords then insert the fresh ones —
   164→   164→        called after a profile edit renames a student."""
   165→   165→        for keyword in self._indexed_keywords.get(user.username, []):
   166→   166→            self.trie.remove(keyword, user.username)
   167→   167→        self._index_user(user)

   247→   247→    def send_request(self, sender: str, receiver: str) -> ConnectionRequest:
   248→   248→        with self._lock:
   249→   249→            if sender == receiver:
   250→   250→                raise ApiError("You cannot connect with yourself.", 400)
   251→   251→            if receiver not in self.users:
   252→   252→                raise ApiError("Student not found.", 404)
   253→   253→            if self.graph.is_connected(sender, receiver):
   254→   254→                raise ApiError("You are already connected with this student.", 409)
   255→   255→            if self.graph.pending_direction(sender, receiver):
   256→   256→                raise ApiError(
   257→   257→                    "There is already a pending request between you two.", 409)
   258→   258→
   259→   259→            created = now_iso()
   260→   260→            request_id = db.insert_request(sender, receiver, created)
   261→   261→            request = ConnectionRequest(request_id, sender, receiver, PENDING, created)
   262→   262→            self.requests[request_id] = request
   263→   263→            self.graph.add_pending(sender, receiver)   # directed edge
   264→   264→            self._notify(receiver, "request_received", sender,
   265→   265→                         f"{self.users[sender].full_name} sent you a "
   266→   266→                         "connection request")
   267→   267→            return request

   269→   269→    def accept_request(self, request_id: int, acting_user: str) -> dict:
   270→   270→        """Accepting converts the directed pending edge into an UNDIRECTED
   271→   271→        connection edge — the core graph transition of the platform."""
   272→   272→        with self._lock:
   273→   273→            request = self._require_pending_request(request_id)
   274→   274→            if request.receiver != acting_user:
   275→   275→                raise ApiError("Only the receiver can accept this request.", 403)
   276→   276→            sender, receiver = request.sender, request.receiver
   277→   277→            if self.graph.is_connected(sender, receiver):
   278→   278→                self._finish_request(request)
   279→   279→                raise ApiError("You are already connected with this student.", 409)
   280→   280→
   281→   281→            self.graph.remove_pending(sender, receiver)   # drop directed edge
   282→   282→            self.graph.add_edge(sender, receiver)         # add undirected edge
   283→   283→            db.insert_connection(sender, receiver, now_iso())
   284→   284→            self._finish_request(request)
   285→   285→            self._notify(sender, "request_accepted", receiver,
   286→   286→                         f"{self.users[receiver].full_name} accepted your "
   287→   287→                         "connection request")
   288→   288→            return {"connected_with": sender}
   289→   289→
   290→   290→    def reject_request(self, request_id: int, acting_user: str) -> dict:
   291→   291→        with self._lock:
   292→   292→            request = self._require_pending_request(request_id)
   293→   293→            if request.receiver != acting_user:
   294→   294→                raise ApiError("Only the receiver can reject this request.", 403)
   295→   295→            self._finish_request(request)     # rejected -> never an edge
   296→   296→            self._notify(request.sender, "request_rejected", acting_user,
   297→   297→                         f"{self.users[acting_user].full_name} declined "
   298→   298→                         "your connection request")
   299→   299→            return {"rejected": request.sender}

   301→   301→    def cancel_request(self, request_id: int, acting_user: str) -> dict:
   302→   302→        with self._lock:
   303→   303→            request = self._require_pending_request(request_id)
   304→   304→            if request.sender != acting_user:
   305→   305→                raise ApiError("Only the sender can cancel this request.", 403)
   306→   306→            self._finish_request(request)
   307→   307→            return {"cancelled": request.receiver}
   308→   308→
   309→   309→    def _require_pending_request(self, request_id) -> ConnectionRequest:
   310→   310→        request = self.requests.get(request_id)
   311→   311→        if request is None or request.status != PENDING:
   312→   312→            raise ApiError("This request no longer exists.", 404)
   313→   313→        return request
   314→   314→
   315→   315→    def _finish_request(self, request: ConnectionRequest) -> None:
   316→   316→        """Terminal state: remove the directed edge everywhere (memory + DB)."""
   317→   317→        self.graph.remove_pending(request.sender, request.receiver)
   318→   318→        self.requests.pop(request.id, None)
   319→   319→        db.delete_request(request.id)

   321→   321→    def remove_connection(self, acting_user: str, other: str) -> dict:
   322→   322→        with self._lock:
   323→   323→            if other not in self.users:
   324→   324→                raise ApiError("Student not found.", 404)
   325→   325→            if not self.graph.is_connected(acting_user, other):

   327→   327→            self.graph.remove_edge(acting_user, other)
   328→   328→            db.delete_connection(acting_user, other)

   364→   364→    def toggle_like(self, post_id: int, acting_user: str) -> dict:
   365→   365→        with self._lock:
   366→   366→            post = self.posts_by_id.get(post_id)
   367→   367→            if post is None:
   368→   368→                raise ApiError("Post not found.", 404)
   369→   369→            likers = self.likes.setdefault(post_id, set())
   370→   370→            if acting_user in likers:
   371→   371→                likers.discard(acting_user)
   372→   372→                db.delete_like(post_id, acting_user)
   373→   373→                liked = False
   374→   374→            else:
   375→   375→                likers.add(acting_user)
   376→   376→                db.insert_like(post_id, acting_user, now_iso())
   377→   377→                liked = True
   378→   378→                if post.author != acting_user:
   379→   379→                    self._notify(post.author, "post_liked", acting_user,
   380→   380→                                 f"{self.users[acting_user].full_name} "
   381→   381→                                 "liked your post")
   382→   382→            return {"liked": liked, "likes": len(likers)}

   407→   407→    def feed(self, username: str) -> List[dict]:
   408→   408→        """News feed = posts by the user and their first-degree connections,
   409→   409→        ordered newest-first through the heap priority queue."""
   410→   410→        with self._lock:
   411→   411→            authors = self.graph.neighbors(username) | {username}
   412→   412→            candidates = [p for p in self.posts if p.author in authors]
   413→   413→            return [self.serialize_post(p, username)
   414→   414→                    for p in self._heap_ordered(candidates)]
   415→   415→
   416→   416→    def posts_by(self, username: str, viewer: str) -> List[dict]:
   417→   417→        with self._lock:
   418→   418→            own = [p for p in self.posts if p.author == username]
   419→   419→            return [self.serialize_post(p, viewer)
   420→   420→                    for p in self._heap_ordered(own)]

   422→   422→    def search(self, query: str, viewer: Optional[str]) -> List[dict]:
   423→   423→        """Prefix search through the Trie + relationship enrichment."""
   424→   424→        with self._lock:
   425→   425→            usernames = self.trie.prefix_matches(query, limit=20)
   426→   426→            results = []
   427→   427→            for username in usernames:
   428→   428→                results.append({
   429→   429→                    **self.user_summary(username),
   430→   430→                    "relationship": self.relationship(viewer, username),
   431→   431→                    "mutual_count": (len(self.graph.mutual(viewer, username))
   432→   432→                                     if viewer else 0),
   433→   433→                })
   434→   434→            return results

   436→   436→    def recommendations(self, viewer: str, limit: int = 10) -> List[dict]:
   437→   437→        """People you may know — Jaccard + mutuals + profile similarity,
   438→   438→        ranked with the bounded top-K heap."""
   439→   439→        with self._lock:
   440→   440→            profiles = {
   441→   441→                user.username: {
   442→   442→                    "programme": user.programme,

   491→   491→    def path_between(self, username: str, other: str) -> dict:
   492→   492→        """Shortest connection chain via BFS (fewest intermediate students)."""
   493→   493→        with self._lock:
   494→   494→            if other not in self.users:
   495→   495→                raise ApiError("Student not found.", 404)
   496→   496→            if other == username:
   497→   497→                raise ApiError("That is your own profile.", 400)
   498→   498→            chain = shortest_path(self.graph, username, other)
   499→   499→            return {
   500→   500→                "source": username,
   501→   501→                "target": other,
   502→   502→                "connected": self.graph.is_connected(username, other),
   503→   503→                "path": chain,
   504→   504→                "distance": (len(chain) - 1) if chain else None,
   505→   505→            }

   568→   568→    def relationship(self, viewer: Optional[str], other: str) -> str:
   569→   569→        """self | connected | pending_out | pending_in | none"""
   570→   570→        if viewer is None:
   571→   571→            return "none"
   572→   572→        if viewer == other:
   573→   573→            return "self"
   574→   574→        if self.graph.is_connected(viewer, other):
   575→   575→            return "connected"
   576→   576→        direction = self.graph.pending_direction(viewer, other)
   577→   577→        if direction == "out":
   578→   578→            return "pending_out"
   579→   579→        if direction == "in":
   580→   580→            return "pending_in"
   581→   581→        return "none"

   630→   630→    def _heap_ordered(self, posts: List[Post]) -> List[Post]:
   631→   631→        """Newest-first ordering via the priority queue.
   632→   632→
   633→   633→        The heap key is -timestamp (a max-heap on time): the smallest key is
   634→   634→        the newest post, so repeated pops yield strictly newest-first order.
   635→   635→        """
   636→   636→        heap = MinHeap.heapify(posts, key=lambda post: -ts_of(post.created_at))
   637→   637→        ordered: List[Post] = []
   638→   638→        while not heap.is_empty():
   639→   639→            ordered.append(heap.pop())
   640→   640→        return ordered
