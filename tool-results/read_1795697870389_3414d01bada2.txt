     1→"""
     2→In-memory network state — the bridge between SQLite and the DSA modules.
     3→
     4→What lives in memory (loaded once at startup)
     5→--------------------------------------------
     6→    users         hash table  username -> User          (O(1) lookups)
     7→    graph         SocialGraph adjacency lists            (accepted + pending edges)
     8→    trie          Trie                                  (prefix search index)
     9→    posts         list + hash table id -> Post
    10→    likes         hash table  post_id -> set(usernames) (O(1) "did I like this?")
    11→    comments      hash table  post_id -> chronological list
    12→    requests      hash table  request_id -> ConnectionRequest (the pending queue)
    13→    notifications hash table  username -> FIFO list
    14→
    15→Every mutation updates the in-memory structures FIRST, then mirrors the
    16→change into SQLite — so the database is pure persistence while ALL
    17→computation (BFS paths, mutuals, feed ordering, recommendations, search)
    18→happens on these in-memory data structures.
    19→
    20→A re-entrant lock guards all operations because the Flask dev server is
    21→multi-threaded.
    22→"""

    50→def ts_of(iso: str) -> float:
    51→    """Parse an ISO timestamp into a Unix float (heap priority key)."""
    52→    try:
    53→        return datetime.fromisoformat(iso).timestamp()
    54→    except (ValueError, TypeError):
    55→        return 0.0

    58→def clean_list(value) -> List[str]:
    59→    """Normalise skills / interests coming from JSON: accepts a list or a
    60→    comma-separated string; deduplicates and trims."""
    61→    if isinstance(value, str):
    62→        parts = value.split(",")
    63→    elif isinstance(value, (list, tuple)):
    64→        parts = [str(item) for item in value]
    65→    else:
    66→        parts = []
    67→    cleaned: List[str] = []
    68→    for part in parts:
    69→        part = part.strip()
    70→        if part and len(part) <= 40 and part not in cleaned:
    71→            cleaned.append(part)
    72→    return cleaned[:12]

    97→    def initialize(self) -> None:
    98→        """Create tables, seed sample data on first boot, load memory."""
    99→        with self._lock:
   100→            db.init_db()
   101→            if not db.fetch_all_users():
   102→                seed.seed_sample_data()
   103→            self.reload_from_db()
   104→
   105→    def reload_from_db(self) -> None:
   106→        """Rebuild every in-memory structure from SQLite. This is the only
   107→        place the database is read in bulk — graph operations afterwards
   108→        never touch SQL."""
   109→        with self._lock:
   110→            self.users = {}
   111→            self.graph = SocialGraph()
   112→            self.trie = Trie()
   113→            self.posts, self.posts_by_id = [], {}
   114→            self.likes, self.comments = {}, {}
   115→            self.comment_likes, self.comments_by_id = {}, {}
   116→            self.requests = {}
   117→            self.notifications = {}
   118→            self._post_counts = {}
   119→            self._indexed_keywords = {}
   120→
   121→            for user in db.fetch_all_users():
   122→                self.users[user.username] = user
   123→                self.graph.add_vertex(user.username)
   124→                self._index_user(user)
   125→
   126→            for user_a, user_b in db.fetch_connections():
   127→                if user_a in self.users and user_b in self.users \
   128→                        and not self.graph.is_connected(user_a, user_b):
   129→                    self.graph.add_edge(user_a, user_b)
   130→
   131→            for request in db.fetch_pending_requests():
   132→                self.requests[request.id] = request
   133→                self.graph.add_pending(request.sender, request.receiver)
   134→
   135→            for post in db.fetch_all_posts():
   136→                self.posts.append(post)
   137→                self.posts_by_id[post.id] = post
   138→                self._post_counts[post.author] = self._post_counts.get(post.author, 0) + 1
   139→
   140→            self.likes = db.fetch_likes()
   141→            self.comments = db.fetch_comments()
   142→            for comment_list in self.comments.values():
   143→                for comment in comment_list:
   144→                    self.comments_by_id[comment.id] = comment
   145→            self.comment_likes = db.fetch_comment_likes()
   146→
   147→            for note in db.fetch_all_notifications():
   148→                self.notifications.setdefault(note.recipient, []).append(note)

   153→    def _index_user(self, user: User) -> None:
   154→        """Insert a student's searchable keywords (username + name words)
   155→        into the Trie."""
   156→        keywords = [user.username] + user.full_name.split()
   157→        for keyword in keywords:
   158→            self.trie.insert(keyword, user.username)
   159→        self._indexed_keywords[user.username] = keywords
   160→
   161→    def _reindex_user(self, user: User) -> None:
   162→        """Remove the old Trie keywords then insert the fresh ones —
   163→        called after a profile edit renames a student."""
   164→        for keyword in self._indexed_keywords.get(user.username, []):
   165→            self.trie.remove(keyword, user.username)
   166→        self._index_user(user)

   171→    def register(self, username, password, full_name, programme, year,
   172→                 bio, skills, interests) -> User:
   173→        with self._lock:
   174→            username = (username or "").strip().lower()
   175→            if not USERNAME_PATTERN.match(username):
   176→                raise ApiError(
   177→                    "Username must be 3-24 characters: lowercase letters, "
   178→                    "digits or underscore.", 400)
   179→            # O(1) duplicate check against the users hash table — the
   180→            # UNIQUE constraint in SQLite is the second line of defence.
   181→            if username in self.users:
   182→                raise ApiError("That username is already taken.", 409)
   183→            if not password or len(password) < PASSWORD_MIN:
   184→                raise ApiError(
   185→                    f"Password must be at least {PASSWORD_MIN} characters.", 400)
   186→            full_name = (full_name or "").strip()
   187→            if not full_name or len(full_name) > 60:
   188→                raise ApiError("Please provide your full name.", 400)
   189→
   190→            user = User(
   191→                username=username,
   192→                full_name=full_name,
   193→                password_hash=hash_password(password),
   194→                programme=(programme or "").strip()[:60],
   195→                year=(year or "").strip()[:20],
   196→                bio=(bio or "").strip()[:280],
   197→                skills=clean_list(skills),
   198→                interests=clean_list(interests),
   199→                created_at=now_iso(),
   200→            )
   201→            db.insert_user(user)              # persist
   202→            self.users[username] = user       # memory hash table
   203→            self.graph.add_vertex(username)    # new graph vertex
   204→            self._index_user(user)            # searchable in the Trie
   205→            return user

   259→    # ====================================================================== #
   260→    # Connection requests (the pending directed edges)
   261→    # ====================================================================== #
   262→    def send_request(self, sender: str, receiver: str) -> ConnectionRequest:
   263→        with self._lock:
   264→            if sender == receiver:
   265→                raise ApiError("You cannot connect with yourself.", 400)
   266→            if receiver not in self.users:
   267→                raise ApiError("Student not found.", 404)
   268→            if self.graph.is_connected(sender, receiver):
   269→                raise ApiError("You are already connected with this student.", 409)
   270→            if self.graph.pending_direction(sender, receiver):
   271→                raise ApiError(
   272→                    "There is already a pending request between you two.", 409)
   273→
   274→            created = now_iso()
   275→            request_id = db.insert_request(sender, receiver, created)
   276→            request = ConnectionRequest(request_id, sender, receiver, PENDING, created)
   277→            self.requests[request_id] = request
   278→            self.graph.add_pending(sender, receiver)   # directed edge
   279→            self._notify(receiver, "request_received", sender,
   280→                         f"{self.users[sender].full_name} sent you a "
   281→                         "connection request")
   282→            return request
   283→
   284→    def accept_request(self, request_id: int, acting_user: str) -> dict:
   285→        """Accepting converts the directed pending edge into an UNDIRECTED
   286→        connection edge — the core graph transition of the platform."""
   287→        with self._lock:
   288→            request = self._require_pending_request(request_id)
   289→            if request.receiver != acting_user:
   290→                raise ApiError("Only the receiver can accept this request.", 403)
   291→            sender, receiver = request.sender, request.receiver
   292→            if self.graph.is_connected(sender, receiver):
   293→                self._finish_request(request)
   294→                raise ApiError("You are already connected with this student.", 409)
   295→
   296→            self.graph.remove_pending(sender, receiver)   # drop directed edge
   297→            self.graph.add_edge(sender, receiver)         # add undirected edge
   298→            db.insert_connection(sender, receiver, now_iso())
   299→            self._finish_request(request)
   300→            self._notify(sender, "request_accepted", receiver,
   301→                         f"{self.users[receiver].full_name} accepted your "
   302→                         "connection request")
   303→            return {"connected_with": sender}
   304→
   305→    def reject_request(self, request_id: int, acting_user: str) -> dict:
   306→        with self._lock:
   307→            request = self._require_pending_request(request_id)
   308→            if request.receiver != acting_user:
   309→                raise ApiError("Only the receiver can reject this request.", 403)
   310→            self._finish_request(request)     # rejected -> never an edge
   311→            self._notify(request.sender, "request_rejected", acting_user,
   312→                         f"{self.users[acting_user].full_name} declined "
   313→                         "your connection request")
   314→            return {"rejected": request.sender}
   315→
   316→    def cancel_request(self, request_id: int, acting_user: str) -> dict:
   317→        with self._lock:
   318→            request = self._require_pending_request(request_id)
   319→            if request.sender != acting_user:
   320→                raise ApiError("Only the sender can cancel this request.", 403)
   321→            self._finish_request(request)
   322→            return {"cancelled": request.receiver}
   323→
   324→    def _require_pending_request(self, request_id) -> ConnectionRequest:
   325→        request = self.requests.get(request_id)
   326→        if request is None or request.status != PENDING:
   327→            raise ApiError("This request no longer exists.", 404)
   328→        return request
   329→
   330→    def _finish_request(self, request: ConnectionRequest) -> None:
   331→        """Terminal state: remove the directed edge everywhere (memory + DB)."""
   332→        self.graph.remove_pending(request.sender, request.receiver)
   333→        self.requests.pop(request.id, None)
   334→        db.delete_request(request.id)
   335→
   336→    def remove_connection(self, acting_user: str, other: str) -> dict:
   337→        with self._lock:
   338→            if other not in self.users:
   339→                raise ApiError("Student not found.", 404)
   340→            if not self.graph.is_connected(acting_user, other):
   341→                raise ApiError("You are not connected with this student.", 409)
   342→            self.graph.remove_edge(acting_user, other)
   343→            db.delete_connection(acting_user, other)
   344→            return {"removed": other}

   349→    def create_post(self, author: str, content: str, media_url: str = "",
   350→                    media_type: str = "") -> Post:
   351→        content = (content or "").strip()
   352→        if not content and not media_url:
   353→            raise ApiError("Post content cannot be empty.", 400)
   354→        if len(content) > POST_MAX_CHARS:
   355→            raise ApiError(f"Posts are limited to {POST_MAX_CHARS} characters.", 400)
   356→        with self._lock:
   357→            created = now_iso()
   358→            post = Post(db.insert_post(author, content, created,
   359→                                       media_url, media_type),
   360→                        author, content, created, media_url, media_type)
   361→            self.posts.append(post)
   362→            self.posts_by_id[post.id] = post
   363→            self._post_counts[author] = self._post_counts.get(author, 0) + 1
   364→            return post
   365→
   366→    def delete_post(self, post_id: int, acting_user: str) -> dict:
   367→        with self._lock:
   368→            post = self.posts_by_id.get(post_id)
   369→            if post is None:
   370→                raise ApiError("Post not found.", 404)
   371→            if post.author != acting_user:
   372→                raise ApiError("You can only delete your own posts.", 403)
   373→            db.delete_post(post_id)          # cascades likes + comments rows
   374→            self.posts.remove(post)
   375→            self.posts_by_id.pop(post_id, None)
   376→            self.likes.pop(post_id, None)
   377→            for comment in self.comments.pop(post_id, []):
   378→                self.comment_likes.pop(comment.id, None)
   379→                self.comments_by_id.pop(comment.id, None)
   380→            self._post_counts[post.author] = self._post_counts.get(post.author, 1) - 1
   381→            return {"deleted": post_id}
   382→
   383→    def toggle_like(self, post_id: int, acting_user: str) -> dict:
   384→        with self._lock:
   385→            post = self.posts_by_id.get(post_id)
   386→            if post is None:
   387→                raise ApiError("Post not found.", 404)
   388→            likers = self.likes.setdefault(post_id, set())
   389→            if acting_user in likers:
   390→                likers.discard(acting_user)
   391→                db.delete_like(post_id, acting_user)
   392→                liked = False
   393→            else:
   394→                likers.add(acting_user)
   395→                db.insert_like(post_id, acting_user, now_iso())
   396→                liked = True
   397→                if post.author != acting_user:
   398→                    self._notify(post.author, "post_liked", acting_user,
   399→                                 f"{self.users[acting_user].full_name} "
   400→                                 "liked your post")
   401→            return {"liked": liked, "likes": len(likers)}

   403→    def add_comment(self, post_id: int, acting_user: str, content: str,
   404→                    parent_id: int | None = None) -> Comment:
   405→        """Add a comment, or — when parent_id is given — a reply to an
   406→        existing comment on the same post (one nesting level)."""
   407→        content = (content or "").strip()
   408→        if not content:
   409→            raise ApiError("Comment cannot be empty.", 400)
   410→        if len(content) > 500:
   411→            raise ApiError("Comments are limited to 500 characters.", 400)
   412→
   413→        parent: Comment | None = None
   414→        if parent_id is not None:
   415→            try:
   416→                parent_id = int(parent_id)
   417→            except (TypeError, ValueError):
   418→                raise ApiError("Invalid reply target.", 400)
   419→
   420→        with self._lock:
   421→            post = self.posts_by_id.get(post_id)
   422→            if post is None:
   423→                raise ApiError("Post not found.", 404)
   424→            if parent_id is not None:
   425→                # Hash-table lookup: the parent comment in O(1).
   426→                parent = self.comments_by_id.get(parent_id)
   427→                if parent is None:
   428→                    raise ApiError("The comment you replied to no longer exists.", 404)
   429→                if parent.post_id != post_id:
   430→                    raise ApiError("You can only reply to comments on the same post.", 400)
   431→                if parent.parent_id is not None:
   432→                    # Keep threads one level deep — treat a reply-to-reply
   433→                    # as a reply to the parent thread's top-level comment.
   434→                    parent_id = parent.parent_id
   435→                    parent = self.comments_by_id[parent_id]
   436→            created = now_iso()
   437→            comment = Comment(db.insert_comment(post_id, acting_user, content,
   438→                                                created, parent_id),
   439→                              post_id, acting_user, content, created, parent_id)
   440→            self.comments.setdefault(post_id, []).append(comment)
   441→            self.comments_by_id[comment.id] = comment
   442→            actor_name = self.users[acting_user].full_name
   443→            if parent is not None:
   444→                if parent.author != acting_user:
   445→                    self._notify(parent.author, "comment_replied", acting_user,
   446→                                 f"{actor_name} replied to your comment")
   447→            elif post.author != acting_user:
   448→                self._notify(post.author, "post_commented", acting_user,
   449→                             f"{actor_name} commented on your post")
   450→            return comment

   462→                db.delete_comment_like(comment_id, acting_user)
   463→                liked = False
   464→            else:
   465→                likers.add(acting_user)
   466→                db.insert_comment_like(comment_id, acting_user, now_iso())
   467→                liked = True
   468→                if comment.author != acting_user:
   469→                    self._notify(comment.author, "comment_liked", acting_user,
   470→                                 f"{self.users[acting_user].full_name} "
   471→                                 "liked your comment")
   472→            return {"liked": liked, "likes": len(likers)}

   475→    # Read models (all computed in memory with the DSA modules)

   477→    def feed(self, username: str) -> List[dict]:
   478→        """News feed = posts by the user and their first-degree connections,
   479→        ordered newest-first through the heap priority queue."""
   480→        with self._lock:
   481→            authors = self.graph.neighbors(username) | {username}
   482→            candidates = [p for p in self.posts if p.author in authors]
   483→            return [self.serialize_post(p, username)
   484→                    for p in self._heap_ordered(candidates)]
   485→
   486→    def posts_by(self, username: str, viewer: str) -> List[dict]:
   487→        with self._lock:
   488→            own = [p for p in self.posts if p.author == username]
   489→            return [self.serialize_post(p, viewer)
   490→                    for p in self._heap_ordered(own)]
   491→
   492→    def search(self, query: str, viewer: Optional[str]) -> List[dict]:
   493→        """Prefix search through the Trie + relationship enrichment."""
   494→        with self._lock:
   495→            usernames = self.trie.prefix_matches(query, limit=20)
   496→            results = []
   497→            for username in usernames:
   498→                results.append({
   499→                    **self.user_summary(username),
   500→                    "relationship": self.relationship(viewer, username),
   501→                    "mutual_count": (len(self.graph.mutual(viewer, username))
   502→                                     if viewer else 0),
   503→                })
   504→            return results
   505→
   506→    def recommendations(self, viewer: str, limit: int = 10) -> List[dict]:
   507→        """People you may know — Jaccard + mutuals + profile similarity,
   508→        ranked with the bounded top-K heap."""
   509→        with self._lock:
   510→            profiles = {
   511→                user.username: {
   512→                    "programme": user.programme,
   513→                    "year": user.year,
   514→                    "skills": user.skills,
   515→                    "interests": user.interests,
   516→                }
   517→                for user in self.users.values()
   518→            }
   519→            ranked = people_you_may_know(self.graph, viewer, profiles, limit)
   520→            enriched = []
   521→            for info in ranked:
   522→                enriched.append({
   523→                    **self.user_summary(info["username"]),
   524→                    **info,
   525→                    "relationship": self.relationship(viewer, info["username"]),
   526→                })
   527→            return enriched

   529→    def connections_of(self, username: str) -> List[dict]:
   530→        with self._lock:
   531→            neighbors = self.graph.neighbors(username)
   532→            return [{
   533→                **self.user_summary(neighbor),
   534→                "mutual_count": len(self.graph.mutual(username, neighbor)),
   535→            } for neighbor in sorted(neighbors)]

   537→    def degrees_of(self, username: str) -> dict:
   538→        """First / second / third-degree network via BFS levels."""
   539→        with self._lock:
   540→            buckets, beyond = connection_degrees(self.graph, username)
   541→            lists = {
   542→                "first": [self.user_summary(u) for u in sorted(buckets[1])][:50],
   543→                "second": [self.user_summary(u) for u in sorted(buckets[2])][:50],
   544→                "third": [self.user_summary(u) for u in sorted(buckets[3])][:50],
   545→            }
   546→            counts = {
   547→                "first": len(buckets[1]),
   548→                "second": len(buckets[2]),
   549→                "third": len(buckets[3]),
   550→                "beyond": len(beyond),
   551→            }
   552→            return {"counts": counts, "lists": lists}

   554→    def mutual_with(self, username: str, other: str) -> List[dict]:
   555→        with self._lock:
   556→            if other not in self.users:
   557→                raise ApiError("Student not found.", 404)
   558→            return [self.user_summary(m)
   559→                    for m in sorted(self.graph.mutual(username, other))]

   561→    def path_between(self, username: str, other: str) -> dict:
   562→        """Shortest connection chain via BFS (fewest intermediate students)."""
   563→        with self._lock:
   564→            if other not in self.users:
   565→                raise ApiError("Student not found.", 404)
   566→            if other == username:
   567→                raise ApiError("That is your own profile.", 400)
   568→            chain = shortest_path(self.graph, username, other)
   569→            return {
   570→                "source": username,
   571→                "target": other,
   572→                "connected": self.graph.is_connected(username, other),
   573→                "path": chain,
   574→                "distance": (len(chain) - 1) if chain else None,
   575→            }

   577→    def requests_for(self, username: str) -> dict:
   578→        """Pending requests received + sent (FIFO order, oldest first)."""
   579→        with self._lock:
   580→            received, sent = [], []
   581→            for request in sorted(self.requests.values(), key=lambda r: r.id):
   582→                if request.receiver == username:
   583→                    received.append({
   584→                        "id": request.id,
   585→                        "created_at": request.created_at,
   586→                        "sender": self.user_summary(request.sender),
   587→                    })
   588→                if request.sender == username:
   589→                    sent.append({
   590→                        "id": request.id,
   591→                        "created_at": request.created_at,
   592→                        "receiver": self.user_summary(request.receiver),
   593→                    })
   594→            return {"received": received, "sent": sent}

   596→    def user_profile(self, username: str, viewer: Optional[str]) -> dict:

   603→                "created_at": user.created_at,
   604→                "relationship": self.relationship(viewer, username),
   605→                "mutual_count": len(mutual),
   606→                "mutual": [self.user_summary(m) for m in sorted(mutual)[:6]],

   638→    def relationship(self, viewer: Optional[str], other: str) -> str:
   639→        """self | connected | pending_out | pending_in | none"""
   640→        if viewer is None:
   641→            return "none"
   642→        if viewer == other:
   643→            return "self"
   644→        if self.graph.is_connected(viewer, other):
   645→            return "connected"
   646→        direction = self.graph.pending_direction(viewer, other)
   647→        if direction == "out":
   648→            return "pending_out"
   649→        if direction == "in":
   650→            return "pending_in"
   651→        return "none"

   669→    def serialize_post(self, post: Post, viewer: Optional[str]) -> dict:
   670→        likers = self.likes.get(post.id, set())
   671→        # Group the flat chronological comment list into threads: top-level
   672→        # comments first, their replies nested one level under each parent.
   673→        flat = self.comments.get(post.id, [])
   674→        children: Dict[int, List[Comment]] = {}
   675→        top_level: List[Comment] = []
   676→        for comment in flat:
   677→            if comment.parent_id is None:
   678→                top_level.append(comment)
   679→            else:
   680→                children.setdefault(comment.parent_id, []).append(comment)
   681→        return {
   682→            "id": post.id,
   683→            "author": self.user_summary(post.author),
   684→            "content": post.content,
   685→            "created_at": post.created_at,
   686→            "media_url": post.media_url,
   687→            "media_type": post.media_type,
   688→            "likes": len(likers),
   689→            "liked_by_me": bool(viewer and viewer in likers),
   690→            "comments": [self.serialize_comment(comment, viewer,
   691→                                                replies=children.get(comment.id, []))
   692→                         for comment in top_level],
   693→            "is_mine": bool(viewer and post.author == viewer),
   694→        }
   695→
   696→    def serialize_comment(self, comment: Comment, viewer: Optional[str],
   697→                          replies: Optional[List[Comment]] = None) -> dict:
   698→        """One comment with its like state (O(1) hash-table lookup) and,
   699→        when supplied, its direct replies nested beneath it."""
   700→        likers = self.comment_likes.get(comment.id, set())
   701→        return {
   702→            "id": comment.id,
   703→            "author": self.user_summary(comment.author),
   704→            "content": comment.content,
   705→            "created_at": comment.created_at,
   706→            "likes": len(likers),
   707→            "liked_by_me": bool(viewer and viewer in likers),
   708→            "parent_id": comment.parent_id,
   709→            "replies": [self.serialize_comment(reply, viewer)
   710→                        for reply in (replies or [])],
   711→        }

   728→    def _heap_ordered(self, posts: List[Post]) -> List[Post]:
   729→        """Newest-first ordering via the priority queue.
   730→
   731→        The heap key is -timestamp (a max-heap on time): the smallest key is
   732→        the newest post, so repeated pops yield strictly newest-first order.
   733→        """
   734→        heap = MinHeap.heapify(posts, key=lambda post: -ts_of(post.created_at))
   735→        ordered: List[Post] = []
   736→        while not heap.is_empty():
   737→            ordered.append(heap.pop())
   738→        return ordered
