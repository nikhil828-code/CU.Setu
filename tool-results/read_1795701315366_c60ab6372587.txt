   100→/** Parse a comma-separated skills string into clean chips. */
   101→function csvToSkills(csv: string): string[] {
   102→  return csv
   103→    .split(',')
   104→    .map((item) => item.trim())
   105→    .filter((item) => item !== '')
   106→}

   109→  return error instanceof ApiError ? error.message : fallback
   110→}

   152→function PostIdeaForm({ onOpenChange }: { onOpenChange: (open: boolean) => void }) {
   153→  const queryClient = useQueryClient()
   154→
   155→  const [title, setTitle] = useState('')
   156→  const [description, setDescription] = useState('')
   157→  const [category, setCategory] = useState<string>('General')
   158→  const [skills, setSkills] = useState('')
   159→  const [teamSize, setTeamSize] = useState(4)
   160→  const [error, setError] = useState<string | null>(null)
   161→
   162→  const skillChips = csvToSkills(skills)
   163→
   164→  const createMutation = useMutation({
   165→    mutationFn: () =>
   166→      apiFetch<CollabItem>('/collabs', {
   167→        method: 'POST',
   168→        body: JSON.stringify({
   169→          title: title.trim(),
   170→          description: description.trim(),
   171→          category,
   172→          skills_needed: skillChips,
   173→          team_size: teamSize,
   174→        }),
   175→      }),
   176→    onSuccess: () => {
   177→      invalidateCollabs(queryClient)
   178→      onOpenChange(false)
   179→      toast({
   180→        title: 'Idea posted',
   181→        description: 'Your project is live — skill-matched teammates can apply to join now.',
   182→      })
   183→    },
   184→    onError: (mutationError) => {
   185→      toast({
   186→        title: 'Could not post idea',
   187→        description: errText(mutationError),
   188→        variant: 'destructive',
   189→      })
   190→    },
   191→  })
   192→
   193→  function handleSubmit(event: React.FormEvent<HTMLFormElement>) {
   194→    event.preventDefault()
   195→    if (createMutation.isPending) return
   196→    if (title.trim().length < 4) {
   197→      setError('Title must be at least 4 characters.')
   198→      return
   199→    }
   200→    setError(null)
   201→    createMutation.mutate()
   202→  }
   203→
   204→  return (
   205→    <form onSubmit={handleSubmit} className="space-y-4">
   206→      <div className="space-y-2">
   207→        <div className="flex items-center justify-between">
   208→          <Label htmlFor="collab-title">Title</Label>
   209→          <span
   210→            className={cn(
   211→              'text-xs tabular-nums',
   212→              title.length > 80 ? 'text-destructive' : 'text-muted-foreground',
   213→            )}
   214→          >
   215→            {title.length}/80
   216→          </span>
   217→        </div>
   218→        <Input
   219→          id="collab-title"
   220→          value={title}
   221→          onChange={(e) => setTitle(e.target.value)}
   222→          maxLength={80}
   223→          placeholder="e.g. Campus Food Review App"
   224→        />
   225→      </div>
   226→
   227→      <div className="space-y-2">
   228→        <Label htmlFor="collab-description">Description</Label>
   229→        <Textarea
   230→          id="collab-description"
   231→          rows={4}
   232→          value={description}
   233→          onChange={(e) => setDescription(e.target.value)}
   234→          maxLength={500}
   235→          placeholder="What are you building, what stage is it at, and what would teammates do?"
   236→        />
   237→      </div>
   238→
   239→      <div className="space-y-2">
   240→        <Label htmlFor="collab-category">Category</Label>
   241→        <Select value={category} onValueChange={setCategory}>
   242→          <SelectTrigger id="collab-category" className="w-full">
   243→            <SelectValue placeholder="Pick a category" />
   244→          </SelectTrigger>
   245→          <SelectContent>
   246→            {CATEGORIES.map((option) => (
   247→              <SelectItem key={option} value={option}>
   248→                {option}
   249→              </SelectItem>
   250→            ))}
   251→          </SelectContent>
   252→        </Select>
   253→      </div>
   254→
   255→      <div className="space-y-2">
   256→        <Label htmlFor="collab-skills">Skills needed</Label>
   257→        <Input
   258→          id="collab-skills"
   259→          value={skills}
   260→          onChange={(e) => setSkills(e.target.value)}
   261→          placeholder="e.g. React, TypeScript, Flask"
   262→        />
   263→        <p className="text-xs text-muted-foreground">Comma-separated</p>
   264→        {skillChips.length > 0 ? (
   265→          <div className="flex flex-wrap gap-1.5">
   266→            {skillChips.map((skill) => (
   267→              <Badge
   268→                key={skill}
   269→                variant="outline"
   270→                className="border-blue-200 bg-blue-50 font-medium text-blue-700"
   271→              >
   272→                {skill}
   273→              </Badge>
   274→            ))}
   275→          </div>
   276→        ) : null}
   277→      </div>
   278→
   279→      <div className="space-y-2">
   280→        <div className="flex items-center justify-between">
   281→          <Label htmlFor="collab-team-size">Team size</Label>
   282→          <span className="text-xs font-semibold tabular-nums text-muted-foreground">
   283→            {teamSize} student{teamSize === 1 ? '' : 's'}
   284→          </span>
   285→        </div>
   286→        <Slider
   287→          id="collab-team-size"
   288→          value={[teamSize]}
   289→          min={TEAM_MIN}
   290→          max={TEAM_MAX}
   291→          step={1}
   292→          onValueChange={(values) => setTeamSize(values[0] ?? teamSize)}
   293→          aria-label="Team size"
   294→        />
   295→      </div>
   296→
   297→      {error ? (
   298→        <p className="text-sm font-medium text-destructive" role="alert">
   299→          {error}
   300→        </p>
   301→      ) : null}
   302→
   303→      <DialogFooter>
   304→        <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   305→          Cancel
   306→        </Button>
   307→        <Button
   308→          type="submit"
   309→          disabled={createMutation.isPending}
   310→          className="bg-gradient-to-r from-amber-500 to-orange-600 text-white shadow-sm hover:from-amber-600 hover:to-orange-700"
   311→        >
   312→          {createMutation.isPending ? (
   313→            <Loader2 className="animate-spin" aria-hidden="true" />
   314→          ) : (
   315→            <Plus aria-hidden="true" />
   316→          )}
   317→          {createMutation.isPending ? 'Posting…' : 'Post idea'}
   318→        </Button>
   319→      </DialogFooter>
   320→    </form>
   321→  )
   322→}

   349→function ApplyForm({ collab, onOpenChange }: { collab: CollabItem; onOpenChange: (open: boolean) => void }) {
   350→  const queryClient = useQueryClient()
   351→  const [message, setMessage] = useState('')
   352→
   353→  const applyMutation = useMutation({
   354→    mutationFn: () =>
   355→      apiFetch<{ ok: boolean; status: string }>(`/collabs/${collab.id}/apply`, {
   356→        method: 'POST',
   357→        body: JSON.stringify({ message: message.trim() }),
   358→      }),
   359→    onSuccess: () => {
   360→      invalidateCollabs(queryClient)
   361→      onOpenChange(false)
   362→      toast({
   363→        title: 'Application sent',
   364→        description: `The owner of "${collab.title}" will review your note.`,
   365→      })
   366→    },
   367→    onError: (error) => {
   368→      toast({
   369→        title: 'Could not apply',
   370→        description: errText(error),
   371→        variant: 'destructive',
   372→      })
   373→    },
   374→  })
   375→
   376→  return (
   377→    <form
   378→      onSubmit={(event) => {
   379→        event.preventDefault()
   380→        if (!applyMutation.isPending) applyMutation.mutate()
   381→      }}
   382→      className="space-y-4"
   383→    >
   384→      <div className="space-y-2">
   385→        <Label htmlFor="apply-message">Message (optional)</Label>
   386→        <Textarea
   387→          id="apply-message"
   388→          rows={3}
   389→          value={message}
   390→          onChange={(e) => setMessage(e.target.value)}
   391→          maxLength={300}
   392→          placeholder="e.g. I've built two React apps and would love to own the frontend."
   393→        />
   394→      </div>
   395→      <DialogFooter>
   396→        <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   397→          Cancel
   398→        </Button>
   399→        <Button
   400→          type="submit"
   401→          disabled={applyMutation.isPending}
   402→          className="bg-gradient-to-r from-amber-500 to-orange-600 text-white shadow-sm hover:from-amber-600 hover:to-orange-700"
   403→        >
   404→          {applyMutation.isPending ? (
   405→            <Loader2 className="animate-spin" aria-hidden="true" />
   406→          ) : (
   407→            <UserPlus aria-hidden="true" />
   408→          )}
   409→          {applyMutation.isPending ? 'Sending…' : 'Send application'}
   410→        </Button>
   411→      </DialogFooter>
   412→    </form>
   413→  )
   414→}

   443→function ApplicationRow({
   444→  application,
   445→  collab,
   446→}: {
   447→  application: CollabApplicationItem
   448→  collab: CollabItem
   449→}) {
   450→  const queryClient = useQueryClient()
   451→  const openProfile = useAppStore((s) => s.openProfile)
   452→
   453→  const acceptMutation = useMutation({
   454→    mutationFn: () =>
   455→      apiFetch<{ ok: boolean; joined: string }>(`/collab-applications/${application.id}/accept`, {
   456→        method: 'POST',
   457→      }),
   458→    onSuccess: (data) => {
   459→      invalidateCollabs(queryClient)
   460→      toast({
   461→        title: 'Application accepted',
   462→        description: `@${data.joined} joined "${collab.title}".`,
   463→      })
   464→    },
   465→    onError: (error) => {
   466→      toast({
   467→        title: 'Could not accept application',
   468→        description: errText(error),
   469→        variant: 'destructive',
   470→      })
   471→    },
   472→  })
   473→
   474→  const rejectMutation = useMutation({
   475→    mutationFn: () =>
   476→      apiFetch<{ ok: boolean; rejected: string }>(`/collab-applications/${application.id}/reject`, {
   477→        method: 'POST',
   478→      }),
   479→    onSuccess: (data) => {
   480→      invalidateCollabs(queryClient)
   481→      toast({
   482→        title: 'Application rejected',
   483→        description: `@${data.rejected} was not added to the team.`,
   484→      })
   485→    },
   486→    onError: (error) => {
   487→      toast({
   488→        title: 'Could not reject application',
   489→        description: errText(error),
   490→        variant: 'destructive',
   491→      })
   492→    },
   493→  })
   494→
   495→  const busy = acceptMutation.isPending || rejectMutation.isPending
   496→
   497→  return (
   498→    <li className="rounded-lg border bg-card p-3">
   499→      <div className="flex items-center gap-3">
   500→        <UserAvatar
   501→          username={application.applicant.username}
   502→          fullName={application.applicant.full_name}
   503→          avatarUrl={application.applicant.avatar_url}
   504→          size="sm"
   505→          onClick={() => openProfile(application.applicant.username)}
   506→        />
   507→        <div className="min-w-0 flex-1">
   508→          <button
   509→            type="button"
   510→            onClick={() => openProfile(application.applicant.username)}
   511→            className="block truncate text-left text-sm font-semibold hover:underline focus-visible:outline-none focus-visible:underline"
   512→          >
   513→            {application.applicant.full_name}
   514→          </button>
   515→          <p className="truncate text-xs text-muted-foreground">
   516→            @{application.applicant.username}
   517→            {application.applicant.programme ? ` · ${application.applicant.programme}` : ''}
   518→          </p>
   519→          <p className="mt-0.5 text-xs text-muted-foreground">
   520→            Applied {timeAgo(application.created_at)}
   521→          </p>
   522→        </div>
   523→      </div>
   524→      {application.message ? (
   525→        <p className="mt-2 border-l-2 border-amber-300 pl-3 text-sm italic text-muted-foreground">
   526→          &ldquo;{application.message}&rdquo;
   527→        </p>
   528→      ) : null}
   529→      <div className="mt-3 flex items-center gap-2">
   530→        <Button
   531→          size="sm"
   532→          className="flex-1 bg-gradient-to-r from-amber-500 to-orange-600 text-white shadow-sm hover:from-amber-600 hover:to-orange-700"
   533→          onClick={() => acceptMutation.mutate()}
   534→          disabled={busy}
   535→        >
   536→          <Check aria-hidden="true" />
   537→          {acceptMutation.isPending ? 'Accepting…' : 'Accept'}
   538→        </Button>
   539→        <Button
   540→          size="sm"
   541→          variant="outline"
   542→          className="flex-1 border-destructive/40 text-destructive hover:bg-destructive/10 hover:text-destructive"
   543→          onClick={() => rejectMutation.mutate()}
   544→          disabled={busy}
   545→        >
   546→          <X aria-hidden="true" />
   547→          {rejectMutation.isPending ? 'Rejecting…' : 'Reject'}
   548→        </Button>
   549→      </div>
   550→    </li>
   551→  )
   552→}

   554→function ApplicationsDialog({
   555→  collab,
   556→  open,
   557→  onOpenChange,
   558→}: {
   559→  collab: CollabItem
   560→  open: boolean
   561→  onOpenChange: (open: boolean) => void
   562→}) {
   563→  const applicationsQuery = useQuery({
   564→    queryKey: ['collab-applications'],
   565→    queryFn: () => apiFetch<{ applications: CollabApplicationItem[] }>('/collab-applications'),
   566→    enabled: open,
   567→  })
   568→
   569→  const applications = (applicationsQuery.data?.applications ?? []).filter(
   570→    (application) => application.collab_id === collab.id,
   571→  )
   572→
   573→  return (
   574→    <Dialog open={open} onOpenChange={onOpenChange}>
   575→      <DialogContent className="sm:max-w-lg">
   576→        <DialogHeader>
   577→          <DialogTitle>Applications — {collab.title}</DialogTitle>
   578→          <DialogDescription>
   579→            Accepting a student adds them to the team immediately.
   580→          </DialogDescription>
   581→        </DialogHeader>
   582→        {applicationsQuery.isLoading ? (
   583→          <div className="space-y-2.5">
   584→            {[0, 1].map((i) => (
   585→              <div key={i} className="rounded-lg border p-3">
   586→                <div className="flex items-center gap-3">
   587→                  <Skeleton className="size-8 rounded-full" />
   588→                  <div className="flex-1 space-y-2">
   589→                    <Skeleton className="h-4 w-36" />
   590→                    <Skeleton className="h-3 w-24" />
   591→                  </div>
   592→                </div>
   593→                <Skeleton className="mt-3 h-8 w-full rounded-md" />
   594→              </div>
   595→            ))}
   596→          </div>
   597→        ) : applications.length === 0 ? (
   598→          <EmptyState
   599→            icon={Inbox}
   600→            title="No pending applications"
   601→            description="When students apply to join this team, their notes will land here."
   602→          />
   603→        ) : (
   604→          <ul className="max-h-[60vh] space-y-2.5 overflow-y-auto thin-scrollbar pr-1">
   605→            {applications.map((application) => (
   606→              <ApplicationRow key={application.id} application={application} collab={collab} />
   607→            ))}
   608→          </ul>
   609→        )}
   610→      </DialogContent>
   611→    </Dialog>
   612→  )
   613→}
