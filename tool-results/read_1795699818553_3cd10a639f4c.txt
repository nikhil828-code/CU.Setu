   110→function levelColor(level: number | null): string | null {

   144→  const [bfsSpeed, setBfsSpeed] = useState(1)

   148→
   149→  /* ---------------- Canvas engine refs ---------------- */
   150→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
   151→  const wrapRef = useRef<HTMLDivElement | null>(null)
   152→  const simNodes = useRef<Map<string, SimNode>>(new Map())

   154→  const adjacency = useRef<Map<string, Set<string>>>(new Map())
   155→  const images = useRef<Map<string, HTMLImageElement>>(new Map())
   156→  const camera = useRef<Camera>({ x: 0, y: 0, zoom: 1 })
   157→  const camTarget = useRef<Camera | null>(null)
   158→  const alpha = useRef(0)
   159→  const hoverNode = useRef<string | null>(null)
   160→  const dragState = useRef<{
   161→    kind: 'node' | 'pan' | null
   162→    node: string | null
   163→    lastX: number
   164→    lastY: number
   165→    downX: number
   166→    downY: number

   168→  }>({ kind: null, node: null, lastX: 0, lastY: 0, downX: 0, downY: 0, moved: 0 })

   170→  const pathRef = useRef<PathPlayback | null>(null)
   171→  const rafRef = useRef<number>(0)
   172→  const sizeRef = useRef({ w: 800, h: 560 })
   173→  const didFit = useRef(false)
   174→  const speedRef = useRef(1)
   175→  const filterRef = useRef<1 | 2 | 3 | null>(null)
   176→  const selectedRef = useRef<string | null>(null)
   177→
   178→  useEffect(() => {
   179→    filterRef.current = degreeFilter
   180→  }, [degreeFilter])
   181→  useEffect(() => {
   182→    selectedRef.current = selected?.username ?? null
   183→  }, [selected])
   184→  useEffect(() => {
   185→    speedRef.current = bfsSpeed
   186→    if (bfsRef.current) bfsRef.current.interval = 620 / bfsSpeed
   187→  }, [bfsSpeed])
   188→
   189→  /* ---------------- Build the simulation when data arrives ---------------- */
   190→  useEffect(() => {
   191→    if (!data || !me) return
   192→    const count = data.nodes.length
   193→    if (count === 0) return
   194→
   195→    // Re-seed only when the node set actually changed.
   196→    const existing = simNodes.current
   197→    const same = new Set([...existing.keys()])
   198→    const incoming = new Set(data.nodes.map((n) => n.username))
   199→    const rebuild = simNodes.current.size === 0 || ![...incoming].every((u) => same.has(u))
   200→
   201→    if (rebuild) {
   202→      const map = new Map<string, SimNode>()
   203→      const R = 260
   204→      data.nodes.forEach((node, i) => {
   205→        const angle = (i / count) * Math.PI * 2 + 0.3
   206→        const jitter = 40 * Math.sin(i * 12.9898)
   207→        map.set(node.username, {
   208→          username: node.username,
   209→          label: node.full_name,
   210→          x: Math.cos(angle) * R + jitter,
   211→          y: Math.sin(angle) * R + jitter,
   212→          vx: 0,
   213→          vy: 0,
   214→          degree: node.connections_count,
   215→          level: node.level,
   216→          avatarUrl: node.avatar_url,
   217→          r: Math.min(26, 15 + node.connections_count * 0.8),
   218→          fixed: false,
   219→        })
   220→        if (node.avatar_url) {
   221→          const img = new Image()
   222→          img.decoding = 'async'
   223→          img.src = `/api${node.avatar_url}${node.avatar_url.includes('?') ? '&' : '?'}XTransformPort=5001`
   224→          img.onload = () => images.current.set(node.username, img)
   225→        }
   226→      })
   227→      simNodes.current = map
   228→      simEdges.current = data.edges.map((e) => [e.source, e.target])
   229→      adjacency.current = new Map(data.nodes.map((n) => [n.username, new Set<string>()]))
   230→      for (const [a, b] of simEdges.current) {
   231→        adjacency.current.get(a)?.add(b)
   232→        adjacency.current.get(b)?.add(a)
   233→      }
   234→      alpha.current = 1
   235→      // Default camera: fit everything.
   236→      fitCamera()
   237→    }
   238→  }, [data, me])
   239→
   240→  const fitCamera = useCallback(() => {
   241→    const nodes = [...simNodes.current.values()]
   242→    if (nodes.length === 0) return
   243→    let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity
   244→    for (const n of nodes) {
   245→      minX = Math.min(minX, n.x); maxX = Math.max(maxX, n.x)
   246→      minY = Math.min(minY, n.y); maxY = Math.max(maxY, n.y)
   247→    }
   248→    const { w, h } = sizeRef.current
   249→    const zoom = Math.min(
   250→      2,
   251→      Math.max(0.35, Math.min(w / (maxX - minX + 160), h / (maxY - minY + 160))),
   252→    )
   253→    camera.current = {
   254→      x: (minX + maxX) / 2,
   255→      y: (minY + maxY) / 2,
   256→      zoom,
   257→    }
   258→  }, [])
   259→
   260→  /* ---------------- Physics + render loop ---------------- */
   261→  useEffect(() => {
   262→    const canvas = canvasRef.current
   263→    if (!canvas) return
   264→    const ctx = canvas.getContext('2d')
   265→    if (!ctx) return
   266→
   267→    const step = (now: number) => {
   268→      const { w, h } = sizeRef.current
   269→      const nodes = simNodes.current
   270→      const arr = [...nodes.values()]
   271→
   272→      // ---- physics (only while the layout is still hot) ----
   273→      if (alpha.current > 0.008 && arr.length > 1) {
   274→        const repulsion = 5200
   275→        for (let i = 0; i < arr.length; i += 1) {
   276→          const a = arr[i]
   277→          for (let j = i + 1; j < arr.length; j += 1) {
   278→            const b = arr[j]
   279→            let dx = b.x - a.x
   280→            let dy = b.y - a.y
   281→            let dist2 = dx * dx + dy * dy
   282→            if (dist2 < 1) {
   283→              dx = Math.random() - 0.5
   284→              dy = Math.random() - 0.5
   285→              dist2 = 1
   286→            }
   287→            const dist = Math.sqrt(dist2)
   288→            const force = repulsion / dist2
   289→            const fx = (dx / dist) * force
   290→            const fy = (dy / dist) * force
   291→            if (!a.fixed) { a.vx -= fx; a.vy -= fy }
   292→            if (!b.fixed) { b.vx += fx; b.vy += fy }
   293→          }
   294→        }
   295→        // springs along edges
   296→        const rest = 120
   297→        for (const [sa, sb] of simEdges.current) {
   298→          const a = nodes.get(sa)
   299→          const b = nodes.get(sb)
   300→          if (!a || !b) continue
   301→          const dx = b.x - a.x
   302→          const dy = b.y - a.y
   303→          const dist = Math.max(1, Math.hypot(dx, dy))
   304→          const force = (dist - rest) * 0.012
   305→          const fx = (dx / dist) * force
   306→          const fy = (dy / dist) * force
   307→          if (!a.fixed) { a.vx += fx; a.vy += fy }
   308→          if (!b.fixed) { b.vx -= fx; b.vy -= fy }
   309→        }
   310→        // gentle centering + integrate
   311→        for (const n of arr) {
   312→          n.vx -= n.x * 0.0006
   313→          n.vy -= n.y * 0.0006
   314→          if (!n.fixed) {
   315→            n.vx *= 0.82
   316→            n.vy *= 0.82
   317→            n.x += Math.max(-18, Math.min(18, n.vx))
   318→            n.y += Math.max(-18, Math.min(18, n.vy))
   319→          } else {
   320→            n.vx = 0
   321→            n.vy = 0
   322→          }
   323→        }
   324→        alpha.current *= 0.985
   325→      }
   326→
   327→      // ---- BFS playback advance ----
   328→      const bfs = bfsRef.current
   329→      if (bfs && bfs.playing && now - bfs.lastTs > bfs.interval) {
   330→        bfs.lastTs = now
   331→        if (bfs.idx < bfs.steps.length - 1) {
   332→          bfs.idx += 1
   333→          const stepData = bfs.steps[bfs.idx]
   334→          bfs.visited.add(stepData.node)
   335→          if (stepData.parent) bfs.discovered.push([stepData.parent, stepData.node])
   336→          setBfsProgress({ idx: bfs.idx, total: bfs.steps.length })
   337→          setBfsCurrent(stepData)
   338→        } else {
   339→          bfs.playing = false
   340→          setBfsRunning(false)
   341→          const last = bfs.steps[bfs.steps.length - 1]
   342→          if (last) setBfsCurrent(last)
   343→        }
   344→      }
   345→
   346→      // ---- path reveal advance ----
   347→      const path = pathRef.current
   348→      if (path && path.revealed < path.edges.length && now - path.lastTs > 420) {
   349→        path.lastTs = now
   350→        path.revealed += 1
   351→      }
   352→
   353→      // ---- smooth camera glide toward a target ----
   354→      if (camTarget.current) {
   355→        const t = camTarget.current
   356→        camera.current.x += (t.x - camera.current.x) * 0.12
   357→        camera.current.y += (t.y - camera.current.y) * 0.12
   358→        camera.current.zoom += (t.zoom - camera.current.zoom) * 0.12
   359→        if (Math.abs(t.x - camera.current.x) < 0.5
   360→            && Math.abs(t.y - camera.current.y) < 0.5
   361→            && Math.abs(t.zoom - camera.current.zoom) < 0.005) {
   362→          camera.current = { ...t }
   363→          camTarget.current = null
   364→        }
   365→      }
   366→
   367→      // ---- render ----
   368→      const cam = camera.current
   369→      const worldToScreenX = (x: number) => (x - cam.x) * cam.zoom + w / 2
   370→      const worldToScreenY = (y: number) => (y - cam.y) * cam.zoom + h / 2
   371→
   372→      // navy backdrop with a soft radial vignette
   373→      ctx.fillStyle = NAVY_BG
   374→      ctx.fillRect(0, 0, w, h)
   375→      const grad = ctx.createRadialGradient(w / 2, h / 2, 60, w / 2, h / 2, Math.max(w, h) * 0.75)
   376→      grad.addColorStop(0, 'rgba(37, 78, 160, 0.20)')
   377→      grad.addColorStop(1, 'rgba(5, 10, 26, 0.55)')
   378→      ctx.fillStyle = grad
   379→      ctx.fillRect(0, 0, w, h)
   380→
   381→      // adjacency sets for highlighting
   382→      const hover = hoverNode.current
   383→      const hoveredNeighbors = hover ? adjacency.current.get(hover) ?? new Set<string>() : null
   384→      const selectedU = selectedRef.current
   385→      const selNeighbors = selectedU ? adjacency.current.get(selectedU) ?? new Set<string>() : null
   386→      const f = filterRef.current
   387→
   388→      // ---------- edges ----------
   389→      for (const [ea, eb] of simEdges.current) {
   390→        const a = nodes.get(ea)
   391→        const b = nodes.get(eb)
   392→        if (!a || !b) continue
   393→        const ax = worldToScreenX(a.x), ay = worldToScreenY(a.y)
   394→        const bx = worldToScreenX(b.x), by = worldToScreenY(b.y)
   395→
   396→        let color = EDGE_COLOR
   397→        let width = 1.2
   398→
   399→        // BFS discovery edge -> glowing cyan
   400→        const bfs2 = bfsRef.current
   401→        if (bfs2 && bfs2.discovered.some(([p, c]) => (p === ea && c === eb) || (p === eb && c === ea))) {
   402→          color = 'rgba(34, 211, 238, 0.75)'
   403→          width = 2.4
   404→        }
   405→        // shortest-path edge -> orange
   406→        const pth = pathRef.current
   407→        if (pth && pth.revealed > 0) {
   408→          const idx = pth.edges.findIndex(([p, c]) => (p === ea && c === eb) || (p === eb && c === ea))
   409→          if (idx !== -1 && idx < pth.revealed) {
   410→            color = 'rgba(251, 146, 60, 0.9)'
   411→            width = 3
   412→          }
   413→        }
   414→        // hover / selection emphasis
   415→        if (hover && (ea === hover || eb === hover)) { color = EDGE_BRIGHT; width = Math.max(width, 2) }
   416→        if (selectedU && (ea === selectedU || eb === selectedU)) {
   417→          if (color === EDGE_COLOR) color = 'rgba(96, 165, 250, 0.35)'
   418→        }
   419→
   420→        ctx.strokeStyle = color
   421→        ctx.lineWidth = width
   422→        ctx.beginPath()
   423→        ctx.moveTo(ax, ay)
   424→        ctx.lineTo(bx, by)
   425→        ctx.stroke()
   426→      }
   427→
   428→      // ---------- nodes ----------
   429→      const bfs2 = bfsRef.current
   430→      const pth = pathRef.current
   431→      const pathSet = pth ? new Set(pth.nodes) : null
   432→
   433→      for (const n of arr) {
   434→        const sx = worldToScreenX(n.x)
   435→        const sy = worldToScreenY(n.y)
   436→        const r = n.r * cam.zoom
   437→        if (sx < -60 || sx > w + 60 || sy < -60 || sy > h + 60) continue
   438→
   439→        const isViewer = n.username === me?.username
   440→        const lvlColor = levelColor(n.level)
   441→        const inBfs = bfs2?.visited.has(n.username) ?? false
   442→        const isBfsCurrent = bfs2 && bfs2.steps[bfs2.idx]?.node === n.username
   443→        const inPath = pathSet?.has(n.username) ?? false
   444→
   445→        // dim nodes not matching the degree filter
   446→        let dimmed = false
   447→        if (f !== null && n.level !== null && n.level > 0) {
   448→          dimmed = n.level !== f && n.level !== 0
   449→        }
   450→
   451→        let ring: string | null = null
   452→        let ringWidth = 2.5
   453→        if (isViewer) { ring = ORANGE; ringWidth = 3 }
   454→        if (lvlColor && !isViewer && f !== null && n.level === f) { ring = lvlColor; ringWidth = 3 }
   455→        if (inPath) { ring = ORANGE; ringWidth = 3 }
   456→        if (inBfs && !ring) { ring = 'rgba(34, 211, 238, 0.8)'; ringWidth = 2 }
   457→        if (n.username === selectedU) { ring = '#FDE68A'; ringWidth = 3.5 }
   458→        if (n.username === hover) { ring = '#E0F2FE'; ringWidth = 2.5 }
   459→
   460→        // glow
   461→        const glowR = r * (isBfsCurrent ? 3.2 : isViewer ? 2.2 : 1.9)
   462→        const glowColor = isViewer ? 'rgba(251,146,60,0.35)'
   463→          : inPath ? 'rgba(251,146,60,0.30)'
   464→          : isBfsCurrent ? 'rgba(34,211,238,0.45)'
   465→          : inBfs ? 'rgba(34,211,238,0.22)'
   466→          : 'rgba(59,130,246,0.30)'
   467→        const g = ctx.createRadialGradient(sx, sy, r * 0.4, sx, sy, glowR)
   468→        g.addColorStop(0, glowColor)
   469→        g.addColorStop(1, 'rgba(10,22,49,0)')
   470→        ctx.fillStyle = g
   471→        ctx.beginPath()
   472→        ctx.arc(sx, sy, glowR, 0, Math.PI * 2)
   473→        ctx.fill()
   474→
   475→        // body: avatar image or gradient fill
   476→        ctx.save()
   477→        ctx.beginPath()
   478→        ctx.arc(sx, sy, r, 0, Math.PI * 2)
   479→        ctx.closePath()
   480→        const img = images.current.get(n.username)
   481→        if (img && img.complete && img.naturalWidth > 0) {
   482→          ctx.clip()
   483→          const size = r * 2
   484→          ctx.drawImage(img, sx - r, sy - r, size, size)
   485→          ctx.restore()
   486→        } else {
   487→          ctx.restore()
   488→          const [c1, c2] = gradientColorsFor(n.username)
   489→          const ng = ctx.createLinearGradient(sx - r, sy - r, sx + r, sy + r)
   490→          ng.addColorStop(0, dimmed ? DIM : c1)
   491→          ng.addColorStop(1, dimmed ? 'rgba(100,116,139,0.3)' : c2)
   492→          ctx.fillStyle = ng
   493→          ctx.beginPath()
   494→          ctx.arc(sx, sy, r, 0, Math.PI * 2)
   495→          ctx.fill()
   496→          // initials
   497→          ctx.fillStyle = dimmed ? 'rgba(203,213,225,0.4)' : 'rgba(255,255,255,0.92)'
   498→          ctx.font = `${Math.max(9, Math.round(r * 0.62))}px ui-sans-serif, system-ui, sans-serif`
   499→          ctx.textAlign = 'center'
   500→          ctx.textBaseline = 'middle'
   501→          const initials = n.label.split(/\s+/).slice(0, 2).map((p) => p.charAt(0).toUpperCase()).join('')
   502→          ctx.fillText(initials, sx, sy + 0.5)
   503→        }
   504→
   505→        if (dimmed) {
   506→          ctx.fillStyle = 'rgba(10,22,49,0.55)'
   507→          ctx.beginPath()
   508→          ctx.arc(sx, sy, r, 0, Math.PI * 2)
   509→          ctx.fill()
   510→        }
   511→
   512→        // ring
