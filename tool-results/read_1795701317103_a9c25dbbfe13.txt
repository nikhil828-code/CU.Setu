   100→   100→/** Parse a comma-separated skills string into clean chips. */
   101→   101→function csvToSkills(csv: string): string[] {
   102→   102→  return csv
   103→   103→    .split(',')
   104→   104→    .map((item) => item.trim())
   105→   105→    .filter((item) => item !== '')
   106→   106→}

   109→   109→  return error instanceof ApiError ? error.message : fallback

   152→   152→function PostIdeaForm({ onOpenChange }: { onOpenChange: (open: boolean) => void }) {
   153→   153→  const queryClient = useQueryClient()
   154→   154→
   155→   155→  const [title, setTitle] = useState('')
   156→   156→  const [description, setDescription] = useState('')
   157→   157→  const [category, setCategory] = useState<string>('General')
   158→   158→  const [skills, setSkills] = useState('')
   159→   159→  const [teamSize, setTeamSize] = useState(4)
   160→   160→  const [error, setError] = useState<string | null>(null)
   161→   161→
   162→   162→  const skillChips = csvToSkills(skills)
   163→   163→
   164→   164→  const createMutation = useMutation({
   165→   165→    mutationFn: () =>
   166→   166→      apiFetch<CollabItem>('/collabs', {
   167→   167→        method: 'POST',
   168→   168→        body: JSON.stringify({
   169→   169→          title: title.trim(),
   170→   170→          description: description.trim(),
   171→   171→          category,
   172→   172→          skills_needed: skillChips,
   173→   173→          team_size: teamSize,
   174→   174→        }),
   175→   175→      }),
   176→   176→    onSuccess: () => {
   177→   177→      invalidateCollabs(queryClient)
   178→   178→      onOpenChange(false)
   179→   179→      toast({
   180→   180→        title: 'Idea posted',
   181→   181→        description: 'Your project is live — skill-matched teammates can apply to join now.',
   182→   182→      })
   183→   183→    },
   184→   184→    onError: (mutationError) => {
   185→   185→      toast({
   186→   186→        title: 'Could not post idea',
   187→   187→        description: errText(mutationError),
   188→   188→        variant: 'destructive',
   189→   189→      })
   190→   190→    },
   191→   191→  })
   192→   192→
   193→   193→  function handleSubmit(event: React.FormEvent<HTMLFormElement>) {
   194→   194→    event.preventDefault()
   195→   195→    if (createMutation.isPending) return
   196→   196→    if (title.trim().length < 4) {
   197→   197→      setError('Title must be at least 4 characters.')
   198→   198→      return
   199→   199→    }
   200→   200→    setError(null)
   201→   201→    createMutation.mutate()
   202→   202→  }
   203→   203→
   204→   204→  return (
   205→   205→    <form onSubmit={handleSubmit} className="space-y-4">
   206→   206→      <div className="space-y-2">
   207→   207→        <div className="flex items-center justify-between">
   208→   208→          <Label htmlFor="collab-title">Title</Label>
   209→   209→          <span
   210→   210→            className={cn(
   211→   211→              'text-xs tabular-nums',
   212→   212→              title.length > 80 ? 'text-destructive' : 'text-muted-foreground',
   213→   213→            )}
   214→   214→          >
   215→   215→            {title.length}/80
   216→   216→          </span>
   217→   217→        </div>
   218→   218→        <Input
   219→   219→          id="collab-title"
   220→   220→          value={title}
   221→   221→          onChange={(e) => setTitle(e.target.value)}
   222→   222→          maxLength={80}
   223→   223→          placeholder="e.g. Campus Food Review App"
   224→   224→        />
   225→   225→      </div>
   226→   226→
   227→   227→      <div className="space-y-2">
   228→   228→        <Label htmlFor="collab-description">Description</Label>
   229→   229→        <Textarea
   230→   230→          id="collab-description"
   231→   231→          rows={4}
   232→   232→          value={description}
   233→   233→          onChange={(e) => setDescription(e.target.value)}
   234→   234→          maxLength={500}
   235→   235→          placeholder="What are you building, what stage is it at, and what would teammates do?"
   236→   236→        />
   237→   237→      </div>
   238→   238→
   239→   239→      <div className="space-y-2">
   240→   240→        <Label htmlFor="collab-category">Category</Label>
   241→   241→        <Select value={category} onValueChange={setCategory}>
   242→   242→          <SelectTrigger id="collab-category" className="w-full">
   243→   243→            <SelectValue placeholder="Pick a category" />
   244→   244→          </SelectTrigger>
   245→   245→          <SelectContent>
   246→   246→            {CATEGORIES.map((option) => (
   247→   247→              <SelectItem key={option} value={option}>
   248→   248→                {option}
   249→   249→              </SelectItem>
   250→   250→            ))}
   251→   251→          </SelectContent>
   252→   252→        </Select>
   253→   253→      </div>
   254→   254→
   255→   255→      <div className="space-y-2">
   256→   256→        <Label htmlFor="collab-skills">Skills needed</Label>
   257→   257→        <Input
   258→   258→          id="collab-skills"
   259→   259→          value={skills}
   260→   260→          onChange={(e) => setSkills(e.target.value)}
   261→   261→          placeholder="e.g. React, TypeScript, Flask"
   262→   262→        />
   263→   263→        <p className="text-xs text-muted-foreground">Comma-separated</p>
   264→   264→        {skillChips.length > 0 ? (
   265→   265→          <div className="flex flex-wrap gap-1.5">
   266→   266→            {skillChips.map((skill) => (
   267→   267→              <Badge
   268→   268→                key={skill}
   269→   269→                variant="outline"
   270→   270→                className="border-blue-200 bg-blue-50 font-medium text-blue-700"
   271→   271→              >
   272→   272→                {skill}
   273→   273→              </Badge>
   274→   274→            ))}
   275→   275→          </div>
   276→   276→        ) : null}
   277→   277→      </div>
   278→   278→
   279→   279→      <div className="space-y-2">
   280→   280→        <div className="flex items-center justify-between">
   281→   281→          <Label htmlFor="collab-team-size">Team size</Label>
   282→   282→          <span className="text-xs font-semibold tabular-nums text-muted-foreground">
   283→   283→            {teamSize} student{teamSize === 1 ? '' : 's'}
   284→   284→          </span>
   285→   285→        </div>
   286→   286→        <Slider
   287→   287→          id="collab-team-size"
   288→   288→          value={[teamSize]}
   289→   289→          min={TEAM_MIN}
   290→   290→          max={TEAM_MAX}
   291→   291→          step={1}
   292→   292→          onValueChange={(values) => setTeamSize(values[0] ?? teamSize)}
   293→   293→          aria-label="Team size"
   294→   294→        />
   295→   295→      </div>
   296→   296→
   297→   297→      {error ? (
   298→   298→        <p className="text-sm font-medium text-destructive" role="alert">
   299→   299→          {error}
   300→   300→        </p>
   301→   301→      ) : null}
   302→   302→
   303→   303→      <DialogFooter>
   304→   304→        <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   305→   305→          Cancel
   306→   306→        </Button>
   307→   307→        <Button
   308→   308→          type="submit"
   309→   309→          disabled={createMutation.isPending}
   310→   310→          className="bg-gradient-to-r from-amber-500 to-orange-600 text-white shadow-sm hover:from-amber-600 hover:to-orange-700"
   311→   311→        >
   312→   312→          {createMutation.isPending ? (
   313→   313→            <Loader2 className="animate-spin" aria-hidden="true" />
   314→   314→          ) : (
   315→   315→            <Plus aria-hidden="true" />
   316→   316→          )}
   317→   317→          {createMutation.isPending ? 'Posting…' : 'Post idea'}
   318→   318→        </Button>
   319→   319→      </DialogFooter>
   320→   320→    </form>
   321→   321→  )
   322→   322→}

   349→   349→function ApplyForm({ collab, onOpenChange }: { collab: CollabItem; onOpenChange: (open: boolean) => void }) {
   350→   350→  const queryClient = useQueryClient()
   351→   351→  const [message, setMessage] = useState('')
   352→   352→
   353→   353→  const applyMutation = useMutation({
   354→   354→    mutationFn: () =>
   355→   355→      apiFetch<{ ok: boolean; status: string }>(`/collabs/${collab.id}/apply`, {
   356→   356→        method: 'POST',
   357→   357→        body: JSON.stringify({ message: message.trim() }),
   358→   358→      }),
   359→   359→    onSuccess: () => {
   360→   360→      invalidateCollabs(queryClient)
   361→   361→      onOpenChange(false)
   362→   362→      toast({
   363→   363→        title: 'Application sent',
   364→   364→        description: `The owner of "${collab.title}" will review your note.`,
   365→   365→      })
   366→   366→    },
   367→   367→    onError: (error) => {
   368→   368→      toast({
   369→   369→        title: 'Could not apply',
   370→   370→        description: errText(error),
   371→   371→        variant: 'destructive',
   372→   372→      })
   373→   373→    },
   374→   374→  })
   375→   375→
   376→   376→  return (
   377→   377→    <form
   378→   378→      onSubmit={(event) => {
   379→   379→        event.preventDefault()
   380→   380→        if (!applyMutation.isPending) applyMutation.mutate()
   381→   381→      }}
   382→   382→      className="space-y-4"
   383→   383→    >
   384→   384→      <div className="space-y-2">
   385→   385→        <Label htmlFor="apply-message">Message (optional)</Label>
   386→   386→        <Textarea
   387→   387→          id="apply-message"
   388→   388→          rows={3}

   390→   390→          onChange={(e) => setMessage(e.target.value)}

   392→   392→          placeholder="e.g. I've built two React apps and would love to own the frontend."
   393→   393→        />
   394→   394→      </div>
   395→   395→      <DialogFooter>
   396→   396→        <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   397→   397→          Cancel
   398→   398→        </Button>
   399→   399→        <Button
   400→   400→          type="submit"
   401→   401→          disabled={applyMutation.isPending}
   402→   402→          className="bg-gradient-to-r from-amber-500 to-orange-600 text-white shadow-sm hover:from-amber-600 hover:to-orange-700"
   403→   403→        >
   404→   404→          {applyMutation.isPending ? (
   405→   405→            <Loader2 className="animate-spin" aria-hidden="true" />
   406→   406→          ) : (
   407→   407→            <UserPlus aria-hidden="true" />
   408→   408→          )}
   409→   409→          {applyMutation.isPending ? 'Sending…' : 'Send application'}
   410→   410→        </Button>
   411→   411→      </DialogFooter>
   412→   412→    </form>
   413→   413→  )
   414→   414→}

   477→   477→        method: 'POST',

   479→   479→    onSuccess: (data) => {
   480→   480→      invalidateCollabs(queryClient)
   481→   481→      toast({
   482→   482→        title: 'Application rejected',
   483→   483→        description: `@${data.rejected} was not added to the team.`,

   486→   486→    onError: (error) => {
   487→   487→      toast({
   488→   488→        title: 'Could not reject application',
   489→   489→        description: errText(error),
   490→   490→        variant: 'destructive',
   491→   491→      })
   492→   492→    },
   493→   493→  })
   494→   494→
   495→   495→  const busy = acceptMutation.isPending || rejectMutation.isPending
   496→   496→
   497→   497→  return (
   498→   498→    <li className="rounded-lg border bg-card p-3">
   499→   499→      <div className="flex items-center gap-3">
   500→   500→        <UserAvatar
   501→   501→          username={application.applicant.username}
   502→   502→          fullName={application.applicant.full_name}
   503→   503→          avatarUrl={application.applicant.avatar_url}
   504→   504→          size="sm"
   505→   505→          onClick={() => openProfile(application.applicant.username)}
   506→   506→        />
   507→   507→        <div className="min-w-0 flex-1">
   508→   508→          <button
